Results are descriptive only. ::: ## Purpose This report reproduces the standard clinical reporting workflow used in regulatory submissions. ```{mermaid} flowchart LR A["SDTM
DM · EX · AE
collected data"] --> B["ADaM
ADSL · ADAE
analysis datasets"] B --> C["QC
automated checks"] C --> D["TLFs
Table 14.1.1 · Table 14.3.1 · Figure 14.3.1"] ``` It uses the open-source [pharmaverse](https://pharmaverse.org) ecosystem in R instead of SAS. Source data are the simulated CDISC pilot study shipped in `pharmaversesdtm` (Xanomeline vs placebo in Alzheimer's disease). No real patient data are used. ## Setup ```{r} library(admiral) library(pharmaversesdtm) library(dplyr) library(tidyr) library(gtsummary) library(ggplot2) dm <- convert_blanks_to_na(pharmaversesdtm::dm) ex <- convert_blanks_to_na(pharmaversesdtm::ex) ae <- convert_blanks_to_na(pharmaversesdtm::ae) ``` ## ADSL: Subject-Level Analysis Dataset One row per subject. Key derivations: | Variable | Meaning | Rule | |---|---|---| | `TRT01P` / `TRT01A` | Planned / actual treatment | From `DM.ARM` / `DM.ACTARM` | | `TRTSDTM` / `TRTSDT` | Treatment start datetime / date | First valid dose in `EX` (dose > 0, or placebo) | | `SAFFL` | Safety population flag | "Y" if the subject received at least one dose | ```{r} ex_ext <- ex %>% derive_vars_dtm(dtc = EXSTDTC, new_vars_prefix = "EXST") adsl <- dm %>% select(-DOMAIN) %>% mutate(TRT01P = ARM, TRT01A = ACTARM) %>% derive_vars_merged( dataset_add = ex_ext, filter_add = (EXDOSE > 0 | (EXDOSE == 0 & grepl("PLACEBO", EXTRT))) & !is.na(EXSTDTM), new_vars = exprs(TRTSDTM = EXSTDTM), order = exprs(EXSTDTM, EXSEQ), mode = "first", by_vars = exprs(STUDYID, USUBJID) ) %>% derive_vars_dtm_to_dt(source_vars = exprs(TRTSDTM)) %>% mutate(SAFFL = if_else(!is.na(TRTSDT), "Y", "N")) count(adsl, TRT01A, SAFFL) ``` ## ADAE: Adverse Events Analysis Dataset One row per adverse event. Key derivations: | Variable | Meaning | Rule | |---|---|---| | `ASTDT` | Analysis start date | Imputed from `AE.AESTDTC` | | `TRTEMFL` | Treatment-emergent flag | "Y" if the event starts on or after the first dose | ```{r} adae <- ae %>% derive_vars_merged( dataset_add = adsl, new_vars = exprs(TRTSDT, TRT01A, SAFFL), by_vars = exprs(STUDYID, USUBJID) ) %>% derive_vars_dt(dtc = AESTDTC, new_vars_prefix = "AST") %>% mutate(TRTEMFL = if_else(!is.na(ASTDT) & ASTDT >= TRTSDT, "Y", NA_character_)) ``` ## Quality checks Basic checks a reviewer would run before producing tables. ```{r} # One row per subject in ADSL stopifnot(!any(duplicated(adsl$USUBJID))) # Every ADAE subject exists in ADSL stopifnot(all(adae$USUBJID %in% adsl$USUBJID)) # Subjects without a treatment start date are not in the safety population stopifnot(all(adsl$SAFFL[is.na(adsl$TRTSDT)] == "N")) ``` ## TLFs ```{r} arm_levels <- c("Placebo", "Xanomeline Low Dose", "Xanomeline High Dose") adsl_saf <- adsl %>% filter(SAFFL == "Y") %>% mutate(TRT01A = factor(TRT01A, levels = arm_levels)) ``` ### Table 14.1.1 Demographic Characteristics – Safety Population ```{r} adsl_saf %>% select(TRT01A, AGE, SEX, RACE) %>% tbl_summary( by = TRT01A, label = list(AGE ~ "Age (years)", SEX ~ "Sex", RACE ~ "Race"), type = list(AGE ~ "continuous2"), statistic = list(AGE ~ c("{mean} ({sd})", "{median} ({p25}, {p75})", "{min}, {max}")), digits = list(AGE ~ c(1, 1, 0, 0, 0, 0, 0)) ) %>% add_overall() ``` No p-values are shown: in a randomised trial, baseline differences between arms are due to chance by design. ### Table 14.3.1 Subjects with Treatment-Emergent Adverse Events by System Organ Class – Safety Population Subjects are counted once per system organ class, regardless of how many events they had. ```{r} n_arm <- count(adsl_saf, TRT01A, name = "N") adae %>% filter(SAFFL == "Y", TRTEMFL == "Y") %>% mutate(TRT01A = factor(TRT01A, levels = arm_levels)) %>% distinct(USUBJID, TRT01A, AEBODSYS) %>% count(AEBODSYS, TRT01A) %>% left_join(n_arm, by = "TRT01A") %>% mutate(value = sprintf("%d (%.1f%%)", n, 100 * n / N)) %>% select(AEBODSYS, TRT01A, value) %>% pivot_wider(names_from = TRT01A, values_from = value, values_fill = "0") %>% arrange(AEBODSYS) %>% knitr::kable(col.names = c("System Organ Class", arm_levels)) ``` ### Figure 14.3.1 Most Frequent Treatment-Emergent Adverse Events by System Organ Class – Safety Population System organ classes reported in at least 15% of subjects in any arm. ```{r} #| fig-alt: "Grouped horizontal bars showing the percentage of subjects with treatment-emergent adverse events by system organ class and treatment arm" ae_pct <- adae %>% filter(SAFFL == "Y", TRTEMFL == "Y") %>% mutate(TRT01A = factor(TRT01A, levels = arm_levels)) %>% distinct(USUBJID, TRT01A, AEBODSYS) %>% count(AEBODSYS, TRT01A) %>% left_join(n_arm, by = "TRT01A") %>% mutate(pct = 100 * n / N) top_soc <- ae_pct %>% group_by(AEBODSYS) %>% filter(max(pct) >= 15) %>% summarise(max_pct = max(pct)) %>% arrange(max_pct) %>% pull(AEBODSYS) ae_pct %>% filter(AEBODSYS %in% top_soc) %>% mutate(AEBODSYS = factor(stringr::str_to_sentence(AEBODSYS), levels = stringr::str_to_sentence(top_soc))) %>% ggplot(aes(x = pct, y = AEBODSYS, fill = TRT01A)) + geom_col(position = position_dodge(width = 0.8), width = 0.75) + scale_fill_manual(values = c("Placebo" = "#9CA3AF", "Xanomeline Low Dose" = "#60A5FA", "Xanomeline High Dose" = "#1D4ED8")) + scale_x_continuous(labels = function(x) paste0(x, "%"), expand = expansion(mult = c(0, 0.05))) + labs(x = "Subjects with at least one TEAE (%)", y = NULL, fill = NULL) + theme_minimal(base_size = 12) + theme(legend.position = "top", panel.grid.major.y = element_blank(), panel.grid.minor = element_blank()) ``` Higher rates in both Xanomeline arms for skin, administration-site and nervous system events, with a dose-dependent pattern for skin and nervous system disorders.
